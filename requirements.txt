#!/usr/bin/env python3
"""
capture_photo.py — Main entry point for photo capture on macOS.

Uses a Chain-of-Responsibility with three strategies:
1. imagesnap (native macOS CLI)
2. OpenCV (VideoCapture)
3. PIL placeholder (solid gray image)

Before capture, performs a camera permission check and ensures dashboard/ directory exists.
"""

import os
import subprocess
import sys
from pathlib import Path

# ----------------------------------------------------------------------
# Fallback capture functions (Strategy pattern encapsulated)
# ----------------------------------------------------------------------

def capture_with_imagesnap(output_path: str) -> bool:
    """Attempt capture using `imagesnap` utility (macOS)."""
    try:
        # imagesnap writes to the given path
        result = subprocess.run(
            ["imagesnap", "-q", output_path],
            capture_output=True,
            timeout=10,
        )
        return result.returncode == 0 and Path(output_path).stat().st_size > 0
    except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
        return False


def capture_with_opencv(output_path: str) -> bool:
    """Attempt capture using OpenCV's VideoCapture (camera 0)."""
    try:
        import cv2
    except ImportError:
        return False

    cap = cv2.VideoCapture(0)
    if not cap.isOpened():
        return False

    try:
        ret, frame = cap.read()
        if ret:
            cv2.imwrite(output_path, frame)
            return True
        return False
    finally:
        cap.release()


def capture_with_placeholder(output_path: str, w: int = 640, h: int = 480) -> bool:
    """Generate a solid grey placeholder image using Pillow."""
    try:
        from PIL import Image
        img = Image.new("RGB", (w, h), (128, 128, 128))
        img.save(output_path)
        return True
    except ImportError:
        return False


# ----------------------------------------------------------------------
# Facade functions: permission check and directory creation
# ----------------------------------------------------------------------

def check_camera_permission() -> bool:
    """
    Probe camera permission by quickly attempting to open camera 0
    and read one frame. Returns True if permission appears granted.
    """
    try:
        import cv2
    except ImportError:
        # If OpenCV not available, can't probe; assume permission is needed later
        return False

    cap = cv2.VideoCapture(0)
    if not cap.isOpened():
        return False

    ret = cap.read()[0]
    cap.release()
    return ret


def ensure_dashboard_dir(directory: str = "dashboard") -> None:
    """Create dashboard directory if it does not exist."""
    Path(directory).mkdir(parents=True, exist_ok=True)


# ----------------------------------------------------------------------
# Main orchestration
# ----------------------------------------------------------------------

def capture_photo(output_dir: str = "dashboard", filename: str = "selfie.jpg") -> None:
    """
    Run the photo capture fallback chain, save result to output_dir/filename.
    Exits with code 0 on success, 1 on failure.
    """
    output_path = os.path.join(output_dir, filename)
    ensure_dashboard_dir(output_dir)

    # Attempt permission check (warning only)
    if not check_camera_permission():
        print("Warning: Camera permission may be denied. Trying anyway...")

    # Chain of responsibility
    if capture_with_imagesnap(output_path):
        print("Photo captured via imagesnap.")
    elif capture_with_opencv(output_path):
        print("Photo captured via OpenCV.")
    elif capture_with_placeholder(output_path):
        print("No camera available – placeholder image saved.")
    else:
        print("Error: All capture methods failed.", file=sys.stderr)
        sys.exit(1)

    print(f"Saved to {output_path}")


if __name__ == "__main__":
    capture_photo()

"""
Unit tests for capture_photo.py fallback logic and file output.
Uses pytest and pytest-mock to mock external dependencies.
"""

import os
import subprocess
from pathlib import Path
from unittest.mock import MagicMock, patch

import pytest

# Import functions under test
from capture_photo import (
    capture_with_imagesnap,
    capture_with_opencv,
    capture_with_placeholder,
    check_camera_permission,
    ensure_dashboard_dir,
    capture_photo,
)


class TestCaptureWithImagesnap:
    def test_success(self, tmp_path, mocker):
        # Create dummy output file to simulate imagesnap success
        out = tmp_path / "test.jpg"
        out.write_text("dummy")
        mocker.patch("subprocess.run", return_value=MagicMock(returncode=0))
        
        result = capture_with_imagesnap(str(out))
        assert result is True

    def test_failure(self, tmp_path, mocker):
        mocker.patch("subprocess.run", side_effect=FileNotFoundError)
        result = capture_with_imagesnap(str(tmp_path / "test.jpg"))
        assert result is False


class TestCaptureWithOpenCv:
    @patch("capture_photo.cv2")
    def test_success(self, mock_cv2, tmp_path):
        mock_cap = MagicMock()
        mock_cap.isOpened.return_value = True
        mock_cap.read.return_value = (True, MagicMock())
        mock_cv2.VideoCapture.return_value = mock_cap

        out = tmp_path / "cv_test.jpg"
        result = capture_with_opencv(str(out))
        assert result is True
        # Check that imwrite was called
        mock_cv2.imwrite.assert_called_once_with(str(out), mock_cap.read()[1])

    @patch("capture_photo.cv2")
    def test_camera_not_opened(self, mock_cv2, tmp_path):
        mock_cap = MagicMock()
        mock_cap.isOpened.return_value = False
        mock_cv2.VideoCapture.return_value = mock_cap

        result = capture_with_opencv(str(tmp_path / "fail.jpg"))
        assert result is False


class TestCaptureWithPlaceholder:
    def test_success(self, tmp_path):
        out = tmp_path / "placeholder.jpg"
        result = capture_with_placeholder(str(out))
        assert result is True
        # Verify file size > 0 (Pillow actually wrote bytes)
        assert Path(out).stat().st_size > 0


class TestPermissionCheck:

        mock_cap.read.return_value = (True, None)
        mock_cv2.VideoCapture.return_value = mock_cap

        assert check_camera_permission() is True

    @patch("capture_photo.cv2")
    def test_denied(self, mock_cv2):
        mock_cap = MagicMock()
        mock_cap.isOpened.return_value = False
        mock_cv2.VideoCapture.return_value = mock_cap

        assert check_camera_permission() is False


class TestEnsureDashboardDir:
    def test_creates_directory(self, tmp_path):
        test_dir = str(tmp_path / "dashboard")
        ensure_dashboard_dir(test_dir)
        assert Path(test_dir).is_dir()

    def test_idempotent(self, tmp_path):
        test_dir = str(tmp_path / "existing_dash")
        Path(test_dir).mkdir()
        ensure_dashboard_dir(test_dir)  # should not raise
        assert Path(test_dir).is_dir()
